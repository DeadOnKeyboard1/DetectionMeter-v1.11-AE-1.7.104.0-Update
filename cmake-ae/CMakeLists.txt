cmake_minimum_required(VERSION 3.21)
project(MaxsuDetectionMeter VERSION 1.1.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MAP_IMPORTED_CONFIG_RELWITHDEBINFO Release)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_SE OFF CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(SKSEPluginAuthor "maxsu")
set(COMMONLIBSSE_SOURCE_DIR "" CACHE PATH "Path to a compatible CommonLibSSE source tree")
set(COMMONLIBSSE_COMPAT_INCLUDE_DIR "" CACHE PATH "Optional compatible CommonLibSSE include override")
set(DKUTIL_INCLUDE_DIR "" CACHE PATH "Path containing the DKUtil include directory")
set(SIMPLEINI_INCLUDE_DIR "" CACHE PATH "Path containing SimpleIni.h")
foreach(variable COMMONLIBSSE_SOURCE_DIR DKUTIL_INCLUDE_DIR SIMPLEINI_INCLUDE_DIR)
 if(NOT ${variable} AND DEFINED ENV{${variable}})
  set(${variable} "$ENV{${variable}}")
 endif()
endforeach()
if(NOT EXISTS "${DKUTIL_INCLUDE_DIR}/DKUtil/Config.hpp")
 message(FATAL_ERROR "Set DKUTIL_INCLUDE_DIR to the directory containing DKUtil")
endif()
if(NOT EXISTS "${SIMPLEINI_INCLUDE_DIR}/SimpleIni.h")
 message(FATAL_ERROR "Set SIMPLEINI_INCLUDE_DIR to the directory containing SimpleIni.h")
endif()
find_package(spdlog CONFIG REQUIRED)
find_package(directxtk CONFIG QUIET)
find_package(CommonLibSSE CONFIG QUIET GLOBAL)
if(NOT TARGET CommonLibSSE::CommonLibSSE)
 if(NOT EXISTS "${COMMONLIBSSE_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Provide an installed CommonLibSSE package or set COMMONLIBSSE_SOURCE_DIR")
 endif()
 add_subdirectory("${COMMONLIBSSE_SOURCE_DIR}" CommonLibSSE)
endif()
if(COMMONLIBSSE_COMPAT_INCLUDE_DIR)
 set_property(TARGET CommonLibSSE::CommonLibSSE PROPERTY INTERFACE_INCLUDE_DIRECTORIES
  "${COMMONLIBSSE_COMPAT_INCLUDE_DIR}")
endif()
configure_file("${ROOT}/cmake/Plugin.h.in" "${CMAKE_CURRENT_BINARY_DIR}/include/Plugin.h" @ONLY)
configure_file("${ROOT}/cmake/version.rc.in" "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
file(GLOB SOURCES "${ROOT}/src/*.cpp")
add_library(MaxsuDetectionMeter SHARED ${SOURCES} "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
target_include_directories(MaxsuDetectionMeter PRIVATE "${ROOT}/src" "${ROOT}/include"
 "${CMAKE_CURRENT_BINARY_DIR}/include" "${DKUTIL_INCLUDE_DIR}" "${SIMPLEINI_INCLUDE_DIR}")
add_library(meter_imgui STATIC "${ROOT}/external/imgui/imgui.cpp" "${ROOT}/external/imgui/imgui_draw.cpp"
 "${ROOT}/external/imgui/imgui_tables.cpp" "${ROOT}/external/imgui/imgui_widgets.cpp"
 "${ROOT}/include/external/imgui_impl_dx11.cpp" "${ROOT}/external/imgui/backends/imgui_impl_win32.cpp")
target_include_directories(meter_imgui PUBLIC "${ROOT}/external/imgui" "${ROOT}/external/imgui/backends")
target_link_libraries(MaxsuDetectionMeter PRIVATE CommonLibSSE::CommonLibSSE spdlog::spdlog meter_imgui d3d11 dxgi d3dcompiler)
target_precompile_headers(MaxsuDetectionMeter PRIVATE "${ROOT}/src/PCH.h")
target_compile_options(MaxsuDetectionMeter PRIVATE /Zi /utf-8 /Zc:preprocessor)
target_link_options(MaxsuDetectionMeter PRIVATE /DEBUG:FULL /INCREMENTAL:NO /MAP)
